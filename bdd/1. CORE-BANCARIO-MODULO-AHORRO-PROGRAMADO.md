// -----------------------------------------------------------------------------
// 0. SELECCIÓN / CREACIÓN DE LA BASE DE DATOS
// -----------------------------------------------------------------------------
use('core_bancario_db');

// Limpieza opcional de colecciones previas en caso de re-ejecución
db.vw_dashboard_ahorro_cliente.drop();
db.vw_cliente_resumen_cuentas.drop();
db.ledger_transacciones.drop();
db.ahorro_programado.drop();
db.cuentas.drop();
db.clientes.drop();


// =============================================================================
// 1. COLECCIÓN: CLIENTES
// Descripción: Almacena la información demográfica y de contacto del cliente.
// Modelo: Embebe la lista de contactos (Email, Teléfonos) en el mismo documento.
// =============================================================================
db.createCollection("clientes", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         required: ["identificacion", "primer_nombre", "primer_apellido", "fecha_nacimiento", "estado"],
         properties: {
            identificacion: { bsonType: "string", description: "Cédula o pasaporte del cliente (Único)" },
            primer_nombre: { bsonType: "string", description: "Primer nombre obligatorios" },
            segundo_nombre: { bsonType: "string" },
            primer_apellido: { bsonType: "string", description: "Primer apellido obligatorio" },
            segundo_apellido: { bsonType: "string" },
            fecha_nacimiento: { bsonType: "date", description: "Fecha de nacimiento para validaciones de edad" },
            estado: { enum: ["ACTIVO", "INACTIVO", "BLOQUEADO"], description: "Estado operativo del cliente" },
            contactos: {
               bsonType: "array",
               description: "Lista de canales de contacto embebidos",
               items: {
                  bsonType: "object",
                  required: ["tipo_contacto", "valor"],
                  properties: {
                     tipo_contacto: { enum: ["EMAIL", "TELEFONO_MOVIL", "TELEFONO_FIJO"] },
                     valor: { bsonType: "string" },
                     es_principal: { bsonType: "bool" }
                  }
               }
            },
            creado_en: { bsonType: "date" }
         }
      }
   }
});

// Índices
db.clientes.createIndex({ "identificacion": 1 }, { unique: true, name: "uk_cliente_identificacion" });


// =============================================================================
// 2. COLECCIÓN: CUENTAS
// Descripción: Cuentas bancarias asociadas al cliente (Ahorros / Corriente).
// =============================================================================
db.createCollection("cuentas", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         required: ["cliente_id", "numero_cuenta", "tipo_cuenta", "saldo", "estado"],
         properties: {
            cliente_id: { bsonType: "objectId", description: "Referencia a _id de la colección clientes" },
            numero_cuenta: { bsonType: "string", description: "Número de cuenta único de 10 dígitos" },
            tipo_cuenta: { enum: ["AHORROS", "CORRIENTE"] },
            saldo: { bsonType: "decimal", minimum: 0, description: "Saldo disponible con precisión Decimal128" },
            estado: { enum: ["ACTIVA", "BLOQUEADA", "CERRADA"] },
            creado_en: { bsonType: "date" }
         }
      }
   }
});

// Índices
db.cuentas.createIndex({ "numero_cuenta": 1 }, { unique: true, name: "uk_cuenta_numero" });
db.cuentas.createIndex({ "cliente_id": 1 }, { name: "idx_cuenta_cliente" });


// =============================================================================
// 3. COLECCIÓN: AHORRO_PROGRAMADO
// Descripción: Planes de ahorro configurados por los clientes.
// Modelo: Unifica la cabecera y el detalle de cuotas/aportes dentro del mismo
// documento mediante el array embebido 'detalles'.
// =============================================================================
db.createCollection("ahorro_programado", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         required: [
            "cuenta_origen_id", "nombre_plan", "modalidad", "monto_objetivo", 
            "es_bloqueado", "tasa_interes_base", "saldo_acumulado", "estado"
         ],
         properties: {
            cuenta_origen_id: { bsonType: "objectId", description: "Referencia a la cuenta de débito" },
            nombre_plan: { bsonType: "string", description: "Ej: Fondo de Emergencia, Vacaciones 2026" },
            modalidad: { enum: ["POR_CUOTA_FIJA", "POR_META"] },
            monto_objetivo: { bsonType: "decimal", minimum: 0 },
            cuota_fija_mensual: { bsonType: ["decimal", "null"] },
            plazo_meses: { bsonType: ["int", "null"] },
            dia_debito_mensual: { bsonType: ["int", "null"], minimum: 1, maximum: 28 },
            es_bloqueado: { bsonType: "bool", description: "Indica si cuenta con blindaje de capital" },
            tasa_interes_base: { bsonType: "decimal" }, // Ej: 4.50%
            tasa_bono_blindaje: { bsonType: "decimal" }, // Ej: 1.10% o 0.00%
            tasa_interes_total: { bsonType: "decimal" }, // Ej: 5.60%
            saldo_acumulado: { bsonType: "decimal", minimum: 0 },
            intereses_devengados_base: { bsonType: "decimal" },
            intereses_devengados_bono: { bsonType: "decimal" },
            cuotas_fallidas_consecutivas: { bsonType: "int", minimum: 0 },
            estado: { 
               enum: ["ACTIVO", "SUSPENDIDO", "COMPLETADO", "CANCELADO_USUARIO", "CANCELADO_INCUMPLIMIENTO"] 
            },
            fecha_inicio: { bsonType: "date" },
            // Subdocumentos: Historial de cuotas y aportes
            detalles: {
               bsonType: "array",
               items: {
                  bsonType: "object",
                  required: ["numero_evento", "tipo_movimiento", "monto", "estado"],
                  properties: {
                     numero_evento: { bsonType: "int" },
                     tipo_movimiento: { enum: ["DEBITO_AUTOMATICO", "APORTE_VOLUNTARIO"] },
                     fecha_programada: { bsonType: ["date", "null"] },
                     fecha_ejecucion: { bsonType: ["date", "null"] },
                     monto: { bsonType: "decimal" },
                     monto_interes_periodo_base: { bsonType: "decimal" },
                     monto_interes_periodo_bono: { bsonType: "decimal" },
                     estado: { enum: ["PENDIENTE", "EXITOSO", "FALLIDO_REINTENTO", "FALLIDO_DEFINITIVO"] },
                     intento_actual: { bsonType: "int", minimum: 0, maximum: 5 },
                     ultimo_mensaje_error: { bsonType: ["string", "null"] }
                  }
               }
            }
         }
      }
   }
});

// Índices
db.ahorro_programado.createIndex({ "cuenta_origen_id": 1 }, { name: "idx_ahorro_cuenta" });
db.ahorro_programado.createIndex({ "estado": 1, "modalidad": 1 }, { name: "idx_ahorro_estado_modalidad" });


// =============================================================================
// 4. COLECCIÓN: LEDGER_TRANSACCIONES (Auditoría Inmutable)
// Descripción: Registro histórico inmutable de todos los débitos, acreditaciones
// e intereses calculados.
// =============================================================================
db.createCollection("ledger_transacciones", {
   validator: {
      $jsonSchema: {
         bsonType: "object",
         required: ["cuenta_id", "plan_id", "tipo_operacion", "monto", "saldo_resultante", "fecha_transaccion"],
         properties: {
            cuenta_id: { bsonType: "objectId" },
            plan_id: { bsonType: "objectId" },
            tipo_operacion: { enum: ["DEBITO_AHORRO", "ABONO_INTERES", "DEVOLUCION_CANCELACION", "PENALIZACION_BONO"] },
            monto: { bsonType: "decimal" },
            saldo_resultante: { bsonType: "decimal" },
            fecha_transaccion: { bsonType: "date" }
         }
      }
   }
});

// Índices
db.ledger_transacciones.createIndex({ "cuenta_id": 1 }, { name: "idx_ledger_cuenta" });
db.ledger_transacciones.createIndex({ "plan_id": 1 }, { name: "idx_ledger_plan" });


// =============================================================================
// 5. VISTA 1: RESUMEN DE CLIENTES Y CUENTAS
// Descripción: Agrupa las cuentas asociadas a cada cliente mediante $lookup.
// =============================================================================
db.createView(
   "vw_cliente_resumen_cuentas",
   "clientes",
   [
      {
         $lookup: {
            from: "cuentas",
            localField: "_id",
            foreignField: "cliente_id",
            as: "cuentas_asociadas"
         }
      },
      {
         $project: {
            _id: 1,
            identificacion: 1,
            nombre_completo: {
               $concat: [
                  "$primer_nombre", " ",
                  { $ifNull: ["$segundo_nombre", ""] }, " ",
                  "$primer_apellido", " ",
                  { $ifNull: ["$segundo_apellido", ""] }
               ]
            },
            estado_cliente: "$estado",
            contactos: "$contactos",
            cuentas: "$cuentas_asociadas"
         }
      }
   ]
);


// =============================================================================
// 6. VISTA 2: DASHBOARD DE AHORRO PROGRAMADO (COMPLETADA)
// Descripción: Genera el consolidador para el frontend, incluyendo el cálculo
// dinámico del porcentaje de avance del objetivo.
// =============================================================================
db.createView(
   "vw_dashboard_ahorro_cliente",
   "ahorro_programado",
   [
      {
         $lookup: {
            from: "cuentas",
            localField: "cuenta_origen_id",
            foreignField: "_id",
            as: "cuenta"
         }
      },
      { $unwind: "$cuenta" },
      {
         $lookup: {
            from: "clientes",
            localField: "cuenta.cliente_id",
            foreignField: "_id",
            as: "cliente"
         }
      },
      { $unwind: "$cliente" },
      {
         $project: {
            _id: 0,
            plan_id: "$_id",
            identificacion: "$cliente.identificacion",
            cliente_nombre: { 
               $concat: ["$cliente.primer_nombre", " ", "$cliente.primer_apellido"] 
            },
            numero_cuenta: "$cuenta.numero_cuenta",
            nombre_plan: 1,
            modalidad: 1,
            monto_objetivo: 1,
            saldo_acumulado: 1,
            intereses_totales_devengados: {
               $add: ["$intereses_devengados_base", "$intereses_devengados_bono"]
            },
            cuota_fija_mensual: 1,
            esta_blindado: "$es_bloqueado",
            tasa_interes_total: 1,
            estado_plan: "$estado",
            // Cálculo del porcentaje de cumplimiento de la meta
            porcentaje_progreso: {
               $cond: [
                  { $eq: ["$monto_objetivo", NumberDecimal("0.00")] },
                  NumberDecimal("0.00"),
                  {
                     $multiply: [
                        { $divide: ["$saldo_acumulado", "$monto_objetivo"] },
                        100
                     ]
                  }
               ]
            }
         }
      }
   ]
);
