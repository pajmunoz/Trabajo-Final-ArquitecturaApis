use('core_bancario_db');

// =============================================================================
// 1. DEFINICIÓN DE LA FUNCIÓN DE TRANSACCIÓN ACID
// =============================================================================
async function registrarAporteLibreMeta(db, client, planIdStr, montoAporteNum) {
   const session = client.startSession();
   session.startTransaction();

   try {
      const planId = new ObjectId(planIdStr);
      const montoAporte = NumberDecimal(montoAporteNum.toString());

      // 1. Obtener y validar el plan de ahorro
      const plan = await db.collection("ahorro_programado").findOne(
         { _id: planId, estado: "ACTIVO", modalidad: "POR_META" },
         { session }
      );

      // 2. Debitar saldo de la cuenta bancaria principal
      if (!plan) {
         throw new Error("El plan de ahorro no existe, no está activo o no es modalidad POR_META.");
      }

      const updateCuenta = await db.collection("cuentas").updateOne(
         { _id: plan.cuenta_origen_id, saldo: { $gte: montoAporte } },
         { $inc: { saldo: NumberDecimal(`-${montoAporteNum}`) } },
         { session }
      );

      if (updateCuenta.modifiedCount === 0) {
         throw new Error("Saldo insuficiente en la cuenta principal para realizar el aporte.");
      }

      const cuentaActual = await db.collection("cuentas").findOne({ _id: plan.cuenta_origen_id }, { session });

      // 3. Calcular interés base del período (4.50% anual mensualizado)
      const tasaBase = parseFloat(plan.tasa_interes_base.toString()) / 100.0;
      const interesBase = NumberDecimal(((montoAporteNum * tasaBase) / 12.0).toFixed(2));

      const siguienteNumeroEvento = (plan.detalles ? plan.detalles.length : 0) + 1;

      // 4. Actualizar saldo acumulado y adjuntar evento en el arreglo embebido 'detalles'
      await db.collection("ahorro_programado").updateOne(
         { _id: planId },
         {
            $inc: {                saldo_acumulado: montoAporte,                intereses_devengados_base: interesBase             },$push: {
               detalles: {
                  numero_evento: siguienteNumeroEvento,
                  tipo_movimiento: "APORTE_VOLUNTARIO",
                  fecha_programada: null,
                  fecha_ejecucion: new Date(),
                  monto: montoAporte,
                  monto_interes_periodo_base: interesBase,
                  monto_interes_periodo_bono: NumberDecimal("0.00"),
                  estado: "EXITOSO",
                  intento_actual: 1,
                  ultimo_mensaje_error: null
               }
            }
         },
         { session }
      );

      // 5. Registrar en el Ledger Inmutable
      await db.collection("ledger_transacciones").insertOne(
         {
            cuenta_id: plan.cuenta_origen_id,
            plan_id: planId,
            tipo_operacion: "DEBITO_AHORRO",
            monto: montoAporte,
            saldo_resultante: cuentaActual.saldo,
            fecha_transaccion: new Date()
         },
         { session }
      );

      await session.commitTransaction();

      return {
         exito: true,
         mensaje: "Aporte registrado exitosamente con atomicidad ACID.",
         plan_id: planIdStr,
         monto_ingresado: montoAporteNum,
         nuevo_saldo_cuenta: cuentaActual.saldo.toString()
      };

   } catch (error) {
      await session.abortTransaction();
      return {
         exito: false,
         mensaje: "Error en la transacción: " + error.message
      };
   } finally {
      session.endSession();
   }
}
