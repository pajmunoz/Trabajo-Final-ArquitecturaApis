workspace "Core Bancario - Contenedores" "Detalle interno del sistema de ahorro programado" {

    model {
        user = person "Cliente Bancario"
        coreLegacy = softwareSystem "Core Bancario Legacy"
        notificationSystem = softwareSystem "Sistema de Notificaciones"

        ahorroSystem = softwareSystem "Sistema de Ahorro Programado" {
            webApp = container "Single Page Application (SPA)" "Interfaz gráfica para el cliente" "React" "Web Browser"
            apiGateway = container "API Gateway" "Punto de entrada único, autenticación JWT, TLS 1.3" "Kong Gateway"
            ahorroService = container "Ahorro Core API" "Gestión de CRUD de planes, simulación pública, vista de clientes" "Spring Boot REST API"
            batchEngine = container "Batch Processor" "Invoca sp_procesar_debitos_ahorro_programado()" "CronJob / Worker"
            database = container "PostgreSQL Database" "Tablas clientes, contactos, cuentas, cabecera, detalle, ledger" "PostgreSQL 15" "Database"
            eventBroker = container "Message Broker" "Publicación/Suscripción de eventos" "RabbitMQ"
        }

        # Relaciones
        user -> webApp "Usa mediante HTTPS"
        webApp -> apiGateway "Realiza peticiones REST (TLS 1.3)"
        apiGateway -> ahorroService "Enruta peticiones autenticadas (OAuth2/JWT)"
        ahorroService -> database "Consulta vistas (vw_cliente_resumen_cuentas) y opera tablas (ACID)"
        batchEngine -> database "Ejecuta procedimiento almacenado de débitos"
        batchEngine -> eventBroker "Publica eventos 'DebitoEjecutado' / 'DebitoFallido'"
        ahorroService -> eventBroker "Publica eventos 'PlanCreado' / 'PlanCancelado'"
        eventBroker -> notificationSystem "Consume eventos y dispara correos/SMS"
        eventBroker -> coreLegacy "Desacopla sincronización con el Core (EDA)"
    }

    views {
        container ahorroSystem "Contenedores" {
            include *
            autolayout lr
        }
    }
}
